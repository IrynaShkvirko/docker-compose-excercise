EXCERCISE 1:

docker run -d --name notes-mongo \
  -e MONGO_INITDB_ROOT_USERNAME=notes \
  -e MONGO_INITDB_ROOT_PASSWORD=notes-password \
  -p 27017:27017 \
  mongo:7

export MONGO_URI='mongodb://notes:notes-password@localhost:27017/?authSource=admin'
export MONGO_DB=notes
export PORT=8080
go run .

curl -s -X POST http://localhost:8080/notes \
  -H 'Content-Type: application/json' \
  -d '{"title":"First note","body":"hello"}'
curl -s http://localhost:8080/notes
curl -s http://localhost:8080/health
