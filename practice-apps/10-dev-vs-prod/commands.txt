EXCERCISE 1:

python3 -m venv .venv
source .venv/bin/activate
pip install -r requirements.txt

APP_ENV=development PORT=5000 python app.py
curl -s http://localhost:5000/config

APP_ENV=production LOG_LEVEL=INFO GREETING='Hello from production' \
  gunicorn -b 0.0.0.0:5000 app:app
curl -s http://localhost:5000/config

EXCERCISE 2:

docker run --rm --name dev-vs-prod-dev \
  -p 5000:5000 \
  -v "$PWD":/app -w /app \
  -e APP_ENV=development -e PORT=5000 \
  python:3.12-slim \
  sh -c "pip install -r requirements.txt && python app.py"

docker run --rm --name dev-vs-prod-prod \
  -p 5001:5000 \
  -v "$PWD":/app -w /app \
  -e APP_ENV=production -e LOG_LEVEL=INFO -e GREETING='Hello from production' \
  python:3.12-slim \
  sh -c "pip install -r requirements.txt && gunicorn -b 0.0.0.0:5000 app:app"

curl -s http://localhost:5000/config
curl -s http://localhost:5001/config
