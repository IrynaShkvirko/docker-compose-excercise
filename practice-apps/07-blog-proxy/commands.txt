EXCERCISE 2:

docker run -d --name blog-nginx \
  -p 8080:80 \
  -e BACKEND_HOST=host.docker.internal \
  -e BACKEND_PORT=5000 \
  -v "$PWD/nginx/default.conf.template:/etc/nginx/templates/default.conf.template:ro" \
  -v "$PWD/frontend:/usr/share/nginx/html:ro" \
  nginx:alpine

curl -s http://localhost:8080/api/health
curl -s -X POST http://localhost:8080/api/posts \
  -H 'Content-Type: application/json' \
  -d '{"title":"Hello","body":"from nginx"}'
