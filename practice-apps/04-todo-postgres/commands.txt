EXCERCISE 1:

docker run -d --name todo-postgres \
  -e POSTGRES_USER=todo \
  -e POSTGRES_PASSWORD=your-password \
  -e POSTGRES_DB=todos \
  -p 5432:5432 \
  -v "$PWD/db/init.sql:/docker-entrypoint-initdb.d/init.sql:ro" \
  postgres:16

docker exec todo-postgres pg_isready -U todo -d todos
docker exec todo-postgres psql -U todo -d todos -c 'SELECT * FROM todos;'

python3 --version
python3 -m venv .venv
source .venv/bin/activate
pip install -r requirements.txt
export DATABASE_URL='postgresql://todo:your-password@localhost:5432/todos'
uvicorn main:app --port 8000
