pip install -r requirements.txt

cat > /tmp/prometheus.local.yml <<'EOF'
global:
  scrape_interval: 5s
scrape_configs:
  - job_name: app
    static_configs:
      - targets: ["host.docker.internal:5000"]
EOF

docker run -d --name metrics-prometheus \
  -p 9090:9090 \
  -v /tmp/prometheus.local.yml:/etc/prometheus/prometheus.yml:ro \
  prom/prometheus
